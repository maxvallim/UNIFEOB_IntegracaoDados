Fase 1 - Importação
Tabela Importação

CREATE TABLE importacao (
    id SERIAL PRIMARY KEY,
    data_importacao TIMESTAMP NOT NULL,
    nome_arquivo VARCHAR(255),
    banco VARCHAR(100),
    agencia VARCHAR(30),
    conta VARCHAR(50),
    data_inicial DATE,
    data_final DATE
);

Tabela Movimento
CREATE TABLE movimento (
    id SERIAL PRIMARY KEY,
    importacao_id INTEGER NOT NULL,
    fitid VARCHAR(100),
    tipo VARCHAR(20),
    data_movimento DATE,
    valor NUMERIC(15,2),
    documento VARCHAR(100),
    historico VARCHAR(255),
    complemento TEXT,
    processado BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (importacao_id)
    REFERENCES importacao(id)
);


Por que essas duas tabelas?
Porque o OFX possui:
1 arquivo
     ↓
N movimentações

Exemplo:
Importação 15
     │
     ├─ PIX
     ├─ Energia
     ├─ TED
     ├─ Aplicação
     └─ Cheque

from sqlalchemy.orm import sessionmaker

load_dotenv()

engine = create_engine(
    DATABASE_URL,
    pool_pre_ping=True
)

SessionLocal = sessionmaker(
    autocommit=False,
    autoflush=False,
    bind=engine
)

app/api/database.py

from fastapi import APIRouter
from sqlalchemy import text

from app.database.database import engine

router = APIRouter(
    prefix="/database",
    tags=["Database"]
)

@router.get("/teste")
def testar_conexao():

    try:

        with engine.connect() as conn:

            resultado = conn.execute(
                text("SELECT NOW()")
            )

            data_hora = resultado.scalar()

            return {
                "status": "sucesso",
                "mensagem": "Conexão com PostgreSQL OK",
                "servidor": str(data_hora)
            }

    except Exception as e:

        return {
            "status": "erro",
            "mensagem": str(e)
        }
