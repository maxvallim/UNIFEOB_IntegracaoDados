
-- CRIAÇÃO DAS TABELAS

CREATE TABLE clientes (
    idClientes SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    cep CHAR(9) NOT NULL
);

CREATE TABLE produtos (
    idProdutos SERIAL PRIMARY KEY,
    descricao VARCHAR(100) NOT NULL,
    unidade CHAR(2) NOT NULL,
    valorUnitario NUMERIC(10,2) NOT NULL
);

CREATE TABLE vendas (
    idVendas SERIAL PRIMARY KEY,
    dataVenda DATE NOT NULL,
    idClientes INTEGER NOT NULL,
    CONSTRAINT fk_vendas_clientes
        FOREIGN KEY (idClientes)
        REFERENCES clientes(idClientes)
);

CREATE TABLE itensVendas (
    idItensVendas SERIAL PRIMARY KEY,
    quantidade NUMERIC(10,2) NOT NULL,
    valorUnitario NUMERIC(10,2) NOT NULL,
    valorTotal NUMERIC(10,2) NOT NULL,
    idVendas INTEGER NOT NULL,
    idProdutos INTEGER NOT NULL,

    CONSTRAINT fk_itens_vendas
        FOREIGN KEY (idVendas)
        REFERENCES vendas(idVendas),

    CONSTRAINT fk_itens_produtos
        FOREIGN KEY (idProdutos)
        REFERENCES produtos(idProdutos)
);

-- ITENS VENDAS

DO $$
DECLARE
    vVenda INTEGER;
    vQtdItens INTEGER;
    vProduto INTEGER;
    vQtd NUMERIC;
    vValor NUMERIC;
BEGIN

    FOR vVenda IN
        SELECT idvendas
        FROM vendas
    LOOP

        vQtdItens := floor(random() * 5 + 1);

        FOR i IN 1..vQtdItens
        LOOP

            SELECT
                idprodutos,
                valorunitario
            INTO
                vProduto,
                vValor
            FROM produtos
            ORDER BY random()
            LIMIT 1;

            vQtd := floor(random() * 4 + 1);

            INSERT INTO itensvendas
            (
                quantidade,
                valorunitario,
                valortotal,
                idvendas,
                idprodutos
            )
            VALUES
            (
                vQtd,
                vValor,
                ROUND(vQtd * vValor,2),
                vVenda,
                vProduto
            );

        END LOOP;

    END LOOP;

END $$;
